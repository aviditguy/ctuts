*** 4. PROJECT: Create an Image Carousel
#+BEGIN_SRC C :includes '(<SDL2/SDL.h> <SDL2/SDL_image.h>) :flags "-lSDL2 -lSDL2_image"
  const int MAX_IMG_LEN = 5;
  int curImg = 0;
  const char *image_path[] = {
    "./project-1/img1.png", "./project-1/img2.png", "./project-1/img3.png",
    "./project-1/img4.jpg", "./project-1/img5.jpg",
  };

  SDL_Init(SDL_INIT_VIDEO);
  IMG_Init(IMG_INIT_PNG | IMG_INIT_JPG);

  SDL_Window *window = SDL_CreateWindow("SDL", SDL_WINDOWPOS_UNDEFINED,
  				      SDL_WINDOWPOS_UNDEFINED, 640, 480, 0);
  SDL_Surface *surface = SDL_GetWindowSurface(window);
  SDL_Surface *images[MAX_IMG_LEN];

  for (int i = 0; i < MAX_IMG_LEN; ++i) {
    SDL_Surface *temp = IMG_Load(image_path[i]);
    images[i] = SDL_ConvertSurface(temp, surface->format, 0);
    SDL_FreeSurface(temp);
   }

  int quit = 0;
  SDL_Event e;
  while (!quit) {
    while (SDL_PollEvent(&e) != 0) {
      if (e.type == SDL_KEYDOWN) {
        switch (e.key.keysym.sym) {
        case SDLK_LEFT:
  	curImg = (curImg == 0) ? (MAX_IMG_LEN - 1) : curImg - 1;
  	break;
        case SDLK_RIGHT:
  	curImg = (curImg == MAX_IMG_LEN - 1) ? 0 : curImg + 1;
  	break;
        case SDLK_ESCAPE:
  	quit = 1;
  	break;
        default:
  	break;
        }
      }
    }
    SDL_Rect stretchimage = {0, 0, 640, 480};
    SDL_BlitScaled(images[curImg], NULL, surface, &stretchimage);
    SDL_UpdateWindowSurface(window);
    SDL_Delay(20);
   }

  for (int i=0; i<MAX_IMG_LEN; ++i)
    SDL_FreeSurface(images[i]);
  SDL_DestroyWindow(window);
  IMG_Quit();
  SDL_Quit();
#+END_SRC

~SDL_Renderer *renderer = SDL_CreateRenderer(window, -1, SDL_RENDERER_ACCELERATED);~
- *Creates a renderer* associated with the window, which is used to draw 2D graphics.
  - *window*: the window to draw to
  - *-1*: lets SDL pick the best available rendering driver
  - *SDL_RENDERER_ACCELERATED*: asks for hardware-accelerated rendering

~SDL_SetRenderDrawColor(renderer, 0xFF, 0xFF, 0xFF, 0xFF);~
- *Sets the drawing color* to white (RGBA: 255, 255, 255, 255).
- This sets the color used for clearing or drawing on the renderer.

~SDL_RenderClear(renderer);~
- Fill the screen with the current drawn color before presenting, in this case WHITE

~SDL_RenderPresent(renderer);~
- *Displays the current rendering* on the screen.
- Since nothing was explicitly drawn yet, only the clear color (white) is shown.

~SDL_DestroyRenderer(renderer);~
- *Cleans up the renderer* and frees associated resources.


#+BEGIN_SRC C :includes '(<SDL2/SDL.h> <SDL2/SDL_image.h>) :flags "-lSDL2 -lSDL2_image"
  SDL_Init(SDL_INIT_VIDEO);
  IMG_Init(IMG_INIT_PNG | IMG_INIT_JPG);

~SDL_Texture *image = SDL_CreateTextureFromSurface(renderer, temp);~
- Converts the surface into a *texture*, which is stored in the GPU for efficient rendering.
- This is necessary to use it with *SDL_RenderCopy()*.

~SDL_FreeSurface(temp);~
- Frees the temporary surface now that the texture is created.
- Avoids memory leaks.

~SDL_RenderCopy(renderer, image, NULL, NULL);~
- Copies the image texture to the renderer.
- The two *NULLs* mean:
  - *source rect* = entire image
  - *destination rect* = fill the whole window

~SDL_RenderPresent(renderer);~
- Updates the window to show the image.

*** Viewports
In *SDL2*, a *viewport* is a rectangular area within the window (or renderer) where drawing operations are applied.
It lets you restrict rendering to a specific region of the screen — like a camera view, sub-window, or split-screen section.
When you set a viewport, *all subsequent rendering is restricted to that area*, and the coordinates are *relative to the viewport*, not the entire window.

*USE CASES*
1. *Split-screen games*
   - Each player’s view is a different viewport.
2. *Mini-maps or UI panels*
   - Draw a map or menu in a fixed area of the screen.
3. *Zoomed or cropped views*
   - Restrict rendering to a region (like a camera).
4. *Picture-in-picture*
   - One part of the screen shows a different scene.

*WHAT THIS PROGRAM DOES?*
- Sets three *viewports* (top-left, top-right, and bottom half of the screen).
- Draws the same image *three times*, each time in a different viewport.
- Displays the result for 5 seconds and then exits.

#+BEGIN_SRC C :includes '(<SDL2/SDL.h> <SDL2/SDL_image.h>) :flags "-lSDL2 -lSDL2_image"
  SDL_Init(SDL_INIT_VIDEO);
  IMG_Init(IMG_INIT_PNG | IMG_INIT_JPG);

  SDL_Window *window = SDL_CreateWindow("SDL", SDL_WINDOWPOS_UNDEFINED,
                                        SDL_WINDOWPOS_UNDEFINED, 640, 480, 0);
  SDL_Renderer *renderer =
      SDL_CreateRenderer(window, -1, SDL_RENDERER_ACCELERATED);

  SDL_Surface *temp = IMG_Load("./project-1/img2.png");
  SDL_Texture *image = SDL_CreateTextureFromSurface(renderer, temp);
  SDL_FreeSurface(temp);

  SDL_Rect topLeft = {0, 0, 320, 240};
  SDL_Rect topRight = {320, 0, 320, 240};
  SDL_Rect bottom = {0, 240, 640, 240};

  SDL_RenderSetViewport(renderer, &topLeft);
  SDL_RenderCopy(renderer, image, NULL, NULL);

  SDL_RenderSetViewport(renderer, &topRight);
  SDL_RenderCopy(renderer, image, NULL, NULL);

  SDL_RenderSetViewport(renderer, &bottom);
  SDL_RenderCopy(renderer, image, NULL, NULL);

  SDL_RenderPresent(renderer);
  SDL_Delay(5000);

  SDL_DestroyTexture(image);
  SDL_DestroyRenderer(renderer);
  SDL_DestroyWindow(window);
  IMG_Quit();
  SDL_Quit();
#+END_SRC

~topLeft~, ~topRight~ and ~bottom~ we define 3 viewports (areas)

~SDL_RenderSetViewport(renderer, &topLeft);~
- Sets rendering to *top-left*.
  
~SDL_RenderCopy(renderer, image, NULL, NULL);~
- Draws the entire image *scaled to fit 320x240* area (area of topLeft).

And similarly, first we set *render viewport* and then we *render* it.


*** Color Keying
*Color keying* is a technique used to make a specific color in an image *transparent*.
In SDL, this is typically used on *surfaces* (not textures directly).

*Color keying works with any surface in SDL, including those created from PNG, BMP, JPG, etc. — as long as the image is loaded into an* ~SDL_Surface~.

*BUT: Why is BMP commonly used with color keying?*
Historically:
- *BMP files* don’t support transparency (no alpha channel).
- So color keying was the only way to make a color "transparent".
- Developers picked a rare color (like cyan or magenta) as a "fake transparency" and keyed it out.

*PNG vs. Color Keying*
- *PNG files* often *already include transparency* via an alpha channel.
- In that case, *you don’t need color keying* — SDL will handle transparency automatically when you create a texture from a PNG with alpha.

*Check if an Image supports transparency (has an alpha channel)?*
- *Load the Image into an SDL_Surface* ~SDL_Surface *surface = IMG_Load("image.png");~
- *Check the pixel format for alpha support* ~if (surface->format->Amask){ printf("Supports Transparency\n"); }~
  - *Amask* stands for *Alpha Mask*
  - If it's *non-zero*, it means the image's pixel format includes an *alpha channel*
  - If *zero*, then there's *no-alpha* — so any transparency must be added manually (e.g., with *color keying*)

- So in this example we will load same image side-by-side in 2 viewport
- In *left viewport* we keep the original image
- In *right viewport* we remove the background color

#+BEGIN_SRC C :includes '(<SDL2/SDL.h> <SDL2/SDL_image.h>) :flags "-lSDL2 -lSDL2_image"
  SDL_Init(SDL_INIT_VIDEO);
  IMG_Init(IMG_INIT_PNG | IMG_INIT_JPG);

  SDL_Window *window = SDL_CreateWindow("SDL", SDL_WINDOWPOS_UNDEFINED,
                                        SDL_WINDOWPOS_UNDEFINED, 640, 480, 0);
  SDL_Renderer *renderer =
      SDL_CreateRenderer(window, -1, SDL_RENDERER_ACCELERATED);

  SDL_SetRenderDrawColor(renderer, 0xFF, 0xFF, 0xFF, 0xFF);
  SDL_RenderClear(renderer);

  SDL_Rect left = {0, 0, 320, 480};
  SDL_Rect right = {320, 0, 320, 480};

  SDL_Surface *temp = IMG_Load("foo.png");

  SDL_RenderSetViewport(renderer, &left);
  SDL_Texture *leftImage = SDL_CreateTextureFromSurface(renderer, temp);
  SDL_RenderCopy(renderer, leftImage, NULL, NULL);

  SDL_RenderSetViewport(renderer, &right);
  SDL_SetColorKey(temp, SDL_TRUE, SDL_MapRGB(temp->format, 0x00, 0xFF, 0xFF));
  SDL_Texture *rightImage = SDL_CreateTextureFromSurface(renderer, temp);
  SDL_RenderCopy(renderer, rightImage, NULL, NULL);

  SDL_RenderPresent(renderer);
  SDL_Delay(3000);

  SDL_FreeSurface(temp);
  SDL_DestroyTexture(leftImage);
  SDL_DestroyTexture(rightImage);
  SDL_DestroyRenderer(renderer);
  SDL_DestroyWindow(window);
  IMG_Quit();
  SDL_Quit();
#+END_SRC


~SDL_SetColorKey(temp, SDL_TRUE, SDL_MapRGB(temp->format, 0x00, 0xFF, 0xFF));~
- *SDL_SetColorKey()* enables color keying for a surface.
- *SDL_TRUE* enables the color key.
- *SDL_MapRGB(...)* converts RGB color *(0,255,255) (cyan)* into the surface's pixel format.
- So, *all pixels in the image that are cyan wil be treated as transparent*.
  
| This must be applied *before converting the surface to a texture*, since textures are GPU-accelerated and don't support color keying directly.

*** Color Modulation
Color Modulation is when you *modify the colors of a texture by a given RGBA color* before rendering.
Think of it like applying a "tint" or "filter" on the texture.
- *Without modulation:*
  - You draw the texture with its original colors.
- *With modulation:*
  - Each pixel ~(r, g, b, a)~ of the texture is multiplied by the modulation color ~(R, G, B, A)~ (values 0–255).

Mathematically, per channel: ~final_channel = (texture_channel * modulation_channel) / 255~

So:
- *(255,255,255)* → no change (white = identity).
- *(255,0,0)* → keeps only the red channel (tints red).
- *(128,128,128)* → darkens (half intensity).
- *(0,0,0)* → black (kills all color).
- Alpha modulation works the same for transparency.


#+BEGIN_SRC C :includes '(<SDL2/SDL.h> <SDL2/SDL_image.h>) :flags "-lSDL2 -lSDL2_image"
  SDL_Init(SDL_INIT_VIDEO);
  IMG_Init(IMG_INIT_PNG | IMG_INIT_JPG);

  SDL_Window *window = SDL_CreateWindow("SDL", SDL_WINDOWPOS_UNDEFINED,
                                        SDL_WINDOWPOS_UNDEFINED, 640, 480, 0);
  SDL_Renderer *renderer =
      SDL_CreateRenderer(window, -1, SDL_RENDERER_ACCELERATED);

  SDL_SetRenderDrawColor(renderer, 0x00, 0x00, 0xFF, 0xFF);
  SDL_RenderClear(renderer);

  SDL_Surface *temp = IMG_Load("colors.png");
  SDL_Texture *image = SDL_CreateTextureFromSurface(renderer, temp);
  SDL_FreeSurface(temp);

  int alpha = 255;
  SDL_SetTextureBlendMode(image, SDL_BLENDMODE_BLEND);
  SDL_SetTextureAlphaMod(image, alpha);

  int quit = 0;
  SDL_Event e;
  while (!quit) {
    while (SDL_PollEvent(&e) != 0) {
      if (e.type == SDL_KEYDOWN) {
        switch (e.key.keysym.sym) {
        case SDLK_UP: alpha += 50; break;
        case SDLK_DOWN: alpha -= 50; break;
        case SDLK_ESCAPE: quit = 1; break;
        default: break;
        }
      }
    }
    alpha = alpha < 0 ? 0 : alpha;
    alpha = alpha > 255 ? 255 : alpha;

    SDL_SetTextureAlphaMod(image, alpha);

    SDL_RenderClear(renderer);
    SDL_RenderCopy(renderer, image, NULL, NULL);
    SDL_RenderPresent(renderer);
  }

  SDL_DestroyTexture(image);
  SDL_DestroyRenderer(renderer);
  SDL_DestroyWindow(window);
  IMG_Quit();
  SDL_Quit();
#+END_SRC
